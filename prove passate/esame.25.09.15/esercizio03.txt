/*
	Input:	m liste ordinate → to_order array contenente m teste di liste
	Output:	array con n elementi ordinati dove n = # elem delle m liste
*/
LIST-ORDER (to_order, m, n)
	BUILD-MIN-HEAP(to_order)
	final ← new array(n)
	for i ← 1 to n // O(n)
		final[i] ← to_order[1].value //min è sempre in testa essendo un min-heap
		to_order[1] ← NEXT(to_order[1])
		if to_order[1] = NIL 
			SWAP(to_order[1], to_order[heapsize[to_order]])
			heapsize[to_order] ← heapsize[to_order] - 1
		MIN-HEAPIFY(to_order, 1) //O(log(m)) perchè dim(to_order) = m
	return final
